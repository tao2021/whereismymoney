cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 70)
endif()
project(my_sgemm LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_executable(sgemm_benchmark bench/benchmark.cpp src/naive.cu src/tiled.cu src/tensor_core.cu)
target_include_directories(sgemm_benchmark PRIVATE src)
target_link_libraries(sgemm_benchmark PRIVATE CUDA::cudart CUDA::cublas)
target_compile_features(sgemm_benchmark PRIVATE cxx_std_17 cuda_std_17)
target_compile_options(sgemm_benchmark PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
enable_testing()
add_test(NAME aligned COMMAND sgemm_benchmark --m 32 --n 48 --k 64 --warmup 1 --repeat 2)
add_test(NAME edges COMMAND sgemm_benchmark --m 37 --n 53 --k 29 --warmup 1 --repeat 2)
add_test(NAME single COMMAND sgemm_benchmark --m 1 --n 1 --k 1 --warmup 1 --repeat 2)
add_test(NAME rectangular COMMAND sgemm_benchmark --m 17 --n 65 --k 257 --warmup 1 --repeat 2)
add_test(NAME invalid COMMAND sgemm_benchmark --m 0)
set_tests_properties(invalid PROPERTIES WILL_FAIL TRUE)
